小学数学题。

首先得知道两条公式吧。

$$A=\pi r^2$$

$$C=2\pi r$$

一个是面积一个是周长。所以一张煎饼暴露的总面积是 $A+Ch$。

整个煎饼塔的表面积是从上往下看的面积再加上所有选中的煎饼的侧面积，也就是最大的那一块煎饼的顶部面积加上所有选中的煎饼的侧面积。

枚举作为最底端的煎饼，然后再选取 $K-1$ 块侧面积最大的煎饼，总面积最大的即为答案。

$\pi$ 要取四舍五入的结果，比如应该为 $3.1415926535898$ 而不是 $3.1415926535897$。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
const double pi=3.1415926535898;
struct Node{
    ll r,h;
    double d,c;
    bool operator >(const Node& N) const{
        return c>N.c;
    }
}s[1145];
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int T;
    cin>>T;
    for(int i=1;i<=T;++i){
        int n,k;
        cin>>n>>k;
        for(int j=1;j<=n;++j){
            cin>>s[j].r>>s[j].h;
            s[j].d=pi*s[j].r*s[j].r;
            s[j].c=2*pi*s[j].r*s[j].h;
        }
        sort(s+1,s+1+n,greater<Node>());
        double ans=0,t;
        for(int j=1;j<=n;++j){
            t=s[j].d+s[j].c;
            int index=1;
            for(int l=1;l<k;++l,++index){
                if(index==j) ++index;
                t+=s[index].c;
            }
            ans=max(ans,t);
        }
        printf("Case #%d: %.9f\n",i,ans);
    }
    return 0;
}
```
