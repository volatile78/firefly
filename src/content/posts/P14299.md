# 思路

以样例一为例。

bfs 和离散化，首先用第一个 bfs 标记所有连通块。

|1  |2  |3  |1  |
|:-:|:-:|:-:|:-:|
|2  |2  |3  |1  |
|1  |2  |3  |1  |
|3  |3  |2  |2  |

会变成

|1  |2  |3  |4  |
|:-:|:-:|:-:|:-:|
|2  |2  |3  |4  |
|5  |2  |3  |4  |
|6  |6  |7  |7  |

然后用第二个 bfs，遍历每一个连通块，可以发现，一个连通块可以获得的价值是连通块的大小加上相邻的最大的一个连通块，注意相邻连通块可能标记不同但是值相同，需要按值处理连通块大小。

# 代码

```cpp
#include<bits/stdc++.h>
#define MAXN 505
#define MAXM 250005
using namespace std;
typedef long long ll;
int H,W;
map<ll,int>lsh;
int a[MAXN][MAXN],b[MAXN][MAXN];
int ltk[MAXM];
bool vis[MAXN][MAXN];
int f[4][2]={{1,0},{-1,0},{0,-1},{0,1}};
int ans;
void bfs(int sx,int sy,int v,int id)
{
    vis[sx][sy]=1;
    b[sx][sy]=id;
    ltk[id]++;
    queue<pair<int,int> >q;
    q.push(make_pair(sx,sy));
    while(!q.empty()){
        for(int i=0;i<4;++i){
            int nx=q.front().first+f[i][0],ny=q.front().second+f[i][1];
            if(nx<1||nx>H||ny<1||ny>W||vis[nx][ny]||a[nx][ny]!=v) continue;
            vis[nx][ny]=1;
            b[nx][ny]=id;
            ltk[id]++;
            q.push(make_pair(nx,ny));
        }
        q.pop();
    }
}
void bfs1(int sx,int sy,int id)
{
    bool vis2[MAXN][MAXN]={false};
    priority_queue<int>pq;
    queue<pair<int,int> >q;
    unordered_set<int>vis1;
    map<int,int>cnt;
    q.push(make_pair(sx,sy));
    vis2[sx][sy]=1;
    while(!q.empty()){
        for(int i=0;i<4;++i){
            int nx=q.front().first+f[i][0],ny=q.front().second+f[i][1];
            if(nx<1||nx>H||ny<1||ny>W||vis2[nx][ny]) continue;
            if(b[nx][ny]==id){
                if(!vis2[nx][ny]){
                    vis2[nx][ny]=1;
                    q.push(make_pair(nx,ny));
                }
                continue;
            }
            if(!vis1.count(b[nx][ny])){
                cnt[a[nx][ny]]+=ltk[b[nx][ny]];
                vis1.insert(b[nx][ny]);
                pq.push(cnt[a[nx][ny]]);
            }
        }
        q.pop();
    }
    if(!pq.empty()) ans=max(ans,ltk[b[sx][sy]]+pq.top());
    else ans=max(ans,ltk[b[sx][sy]]);
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int t=0,t1=0;
    cin>>H>>W;
    for(int i=1;i<=H;++i){
        for(int j=1;j<=W;++j){
            cin>>a[i][j];
            if(!lsh.count(a[i][j])){
                t++;
                lsh[a[i][j]]=t;
            }
            a[i][j]=lsh[a[i][j]];
        }
    }
    for(int i=1;i<=H;++i){
        for(int j=1;j<=W;++j){
            if(!vis[i][j]){
                t1++;
                bfs(i,j,a[i][j],t1);
            }
        }
    }
    vector<bool>vt(t1+1,false);
    for(int i=1;i<=H;++i){
        for(int j=1;j<=W;++j){
            if(!vt[b[i][j]]){
                vt[b[i][j]]=1;
                bfs1(i,j,b[i][j]);
            }
        }
    }
    cout<<ans;
    return 0;
}
```
