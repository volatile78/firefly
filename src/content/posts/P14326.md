# 思路

bfs，只要判断当前的位置是否与上一个位置的颜色不同，就可以把达到当前位置所需的步数更新为上一个位置的步数加一。

注意初始化等问题。

```cpp
#include<bits/stdc++.h>
#define MAXN 505
using namespace std;
typedef long long ll;
int ans=-1;
int H,W;
bool a[MAXN][MAXN];
int f[4][2]={{1,0},{-1,0},{0,1},{0,-1}};
int b[MAXN][MAXN];
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>H>>W;
    for(int i=1;i<=H;++i){
        for(int j=1;j<=W;++j){
            char ch;
            cin>>ch;
            if(ch=='#') a[i][j]=1;
            b[i][j]=-1;
        }
    }
    b[1][1]=0;
    queue<pair<int,int> >q;
    q.push(make_pair(1,1));
    while(!q.empty()){
        auto [x,y]=q.front();
        q.pop();
        if(x==H&&y==W){
            ans=b[x][y];
            break;
        }
        for(int i=0;i<4;i++){
            int nx=x+f[i][0],ny=y+f[i][1];
            if(nx<1||nx>H||ny<1||ny>W||b[nx][ny]!=-1||a[nx][ny]==a[x][y]) continue;
            b[nx][ny]=b[x][y]+1;
            q.push(make_pair(nx,ny));
        }
    }
    cout<<ans;
    return 0;
}
```
