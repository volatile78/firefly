**不要遍历边框。**

先定位左上角，如果发现是课程体系框的左上角，就进入函数。

```cpp
if(nt[i][j]=='+'&&nt[i][j+1]=='-'&&nt[i-1][j]!='|') addClass(i,j,++sum);
```

函数内遍历每一行，注意去掉空格不分大小写，存到数组里。

```cpp
void addClass(int x,int y,int t)
{
    int z;
    for(int i=x+1;;++i){
        if(nt[i][y]=='+'){
            z=i;
            break;
        }
    }
    for(int i=x+1;i<z;++i){
        string clsnm="";
        for(int j=y+1;nt[i][j]!='|';++j){
            if(nt[i][j]==' ') continue;
            if(nt[i][j]>='A'&&nt[i][j]<='Z') nt[i][j]=char(nt[i][j]-'A'+'a');
            clsnm+=nt[i][j];
        }
        if(clsnm!=""){
            if(course.count(clsnm)) zs[t].push_back(course[clsnm]);
            else{
                ++cs;
                course[clsnm]=cs;
                zs[t].push_back(cs);
            }
        }
    }
}
```

最后暴力求最小就行了。

```cpp line-numbers
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
map<string,int>course;
vector<int>zs[55];
string nt[405];
int cs;
void addClass(int x,int y,int t)
{
    int z;
    for(int i=x+1;;++i){
        if(nt[i][y]=='+'){
            z=i;
            break;
        }
    }
    for(int i=x+1;i<z;++i){
        string clsnm="";
        for(int j=y+1;nt[i][j]!='|';++j){
            if(nt[i][j]==' ') continue;
            if(nt[i][j]>='A'&&nt[i][j]<='Z') nt[i][j]=char(nt[i][j]-'A'+'a');
            clsnm+=nt[i][j];
        }
        if(clsnm!=""){
            if(course.count(clsnm)) zs[t].push_back(course[clsnm]);
            else{
                ++cs;
                course[clsnm]=cs;
                zs[t].push_back(cs);
            }
        }
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int sum=0,s=0,ans=INT_MAX;
    string str;
    while(getline(cin,str)){
        nt[s]=str;
        ++s;
    }
    for(int i=1;i<s-1;++i){
        for(int j=1;j<nt[0].size()-1;++j){
            if(nt[i][j]=='+'&&nt[i][j+1]=='-'&&nt[i-1][j]!='|') addClass(i,j,++sum);
        }
    }
    for(int i=1;i<sum-1;++i){
        for(int j=i+1;j<sum;++j){
            for(int k=j+1;k<=sum;++k){
                set<int>t;
                for(auto l:zs[i]) t.insert(l);
                for(auto l:zs[j]) t.insert(l);
                for(auto l:zs[k]) t.insert(l);
                ans=min<int>(ans,t.size());
            }
        }
    }
    cout<<ans;
    return 0;
}
```
