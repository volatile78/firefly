# 题意

给你一个棋盘，如果棋盘上为 $\texttt{O}$，则一定有船；如果棋盘上为 $\texttt{X}$，则一定没船。船不可重叠，告诉你所有船的长度，求船的排列方案数。

# 思路

样例比较小用 dfs 暴力求解即可，在添加答案时需要检查方案是否合法（所有的 $\texttt{O}$ 上必须有船）。

有一个坑就是长度为 $1$ 的船在横着放和竖着放时都一样。

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
char qp[6][6];
bool mp[6][6];
int n,k,ship[6],ans=0;
bool check()
{
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j){
            if(qp[i][j]=='O'&&!mp[i][j]) return 0;
        }
    }
    return 1;
}
void dfs(int in)
{
    if(in==k+1){
        if(check()) ans++;
        return;
    }
    int len=ship[in];
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j){
            if(j+len-1<=n){
                bool flag=1;
                for(int t=j;t<j+len;++t){
                    if(qp[i][t]=='X'||mp[i][t]){
                        flag=0;
                        break;
                    }
                }
                if(flag){
                    for(int t=j;t<j+len;++t) mp[i][t]=1;
                    dfs(in+1);
                    for(int t=j;t<j+len;++t) mp[i][t]=0;
                }
            }
            if(i+len-1<=n&&len!=1){
                bool flag=1;
                for(int t=i;t<i+len;++t){
                    if(qp[t][j]=='X'||mp[t][j]){
                        flag=0;
                        break;
                    }
                }
                if(flag){
                    for(int t=i;t<i+len;++t) mp[t][j]=1;
                    dfs(in+1);
                    for(int t=i;t<i+len;++t) mp[t][j]=0;
                }
            }
        }
    }
}
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n>>k;
    for(int i=1;i<=n;++i){
        for(int j=1;j<=n;++j) cin>>qp[i][j];
    }
    for(int i=1;i<=k;++i) cin>>ship[i];
    dfs(1);
    cout<<ans;
    return 0;
}
```
