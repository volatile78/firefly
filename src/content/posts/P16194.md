对于每一行输入，截取它的数字部分，由于它的数字只会有一段，所以直接用字符串拼贴。

```cpp
string s,id="";
getline(cin,s);//有空格
for(int i=0;i<s.size();++i){
    if(s[i]>='0'&&s[i]<='9') id+=s[i];
}
```

用一个 map 存每个学校的人数，如果这个学校是第一次出现，那么统计答案的变量 $t$ 加一。如果这个学校的人数超过 $5$ 人，那么答案 $t$ 减一，并且要把这个学校的人数设成一个很小的数避免重复计算。

最后遍历输出所有人数在 $1$ 到 $5$ 之间的学校编号即可。

```cpp
#include<bits/stdc++.h>
#define endl '\n'
using namespace std;
typedef long long ll;
unordered_map<string,int>sc;
int main()
{
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int n,t=0;
    cin>>n;
    cin.ignore();
    while(n--){
        string s,id="";
        getline(cin,s);
        for(int i=0;i<s.size();++i){
            if(s[i]>='0'&&s[i]<='9') id+=s[i];
        }
        if(!sc.count(id)) ++t;
        ++sc[id];
        if(sc[id]>5){
            --t;
            sc[id]=-1145;
        }
    }
    cout<<t<<endl;
    for(auto it:sc){
        if(it.second<=5&&it.second>=1) cout<<it.first<<endl;
    }
    return 0;
}
```
